import boto3
import os
import re
import time
import json
from collections import defaultdict
from datetime import datetime

logs = boto3.client('logs')
sns = boto3.client('sns')

SNS_TOPIC_ARN = os.getenv(
    "SNS_TOPIC_ARN"
)

if not SNS_TOPIC_ARN:
    raise Exception(
        "SNS_TOPIC_ARN environment variable not configured"
    )

USER_REGEX = r'user "([^"]+)"'
IP_REGEX = r'UTC:([0-9]+\.[0-9]+\.[0-9]+\.[0-9]+)'


def lambda_handler(event, context):

    print("Received Event:")
    print(json.dumps(event))

    try:

        alarm_state = event["alarmData"]["state"]["value"]

        if alarm_state != "ALARM":

            print("Alarm is not in ALARM state")

            return {
                "statusCode": 200
            }

    except Exception as e:

        print(
            f"Unable to determine alarm state: {e}"
        )

        return {
            "statusCode": 200
        }

    alarm_name = event["alarmData"]["alarmName"]

    # ---------------------------------------
    # Alarm Details
    # ---------------------------------------

    reason_data = json.loads(
        event["alarmData"]["state"]["reasonData"]
    )

    threshold = reason_data.get(
        "threshold"
    )

    period = reason_data.get(
        "period"
    )

    evaluated_datapoints = reason_data.get(
        "evaluatedDatapoints",
        []
    )

    datapoints_alarm = len(
        evaluated_datapoints
    )

    # ---------------------------------------
    # Metric Name
    # ---------------------------------------

    metric_name = (
        event["alarmData"]["configuration"]
        ["metrics"][0]
        ["metricStat"]
        ["metric"]["name"]
    )

    print(
        f"Metric Name: {metric_name}"
    )

    # ---------------------------------------
    # Derive Cluster Name
    # ---------------------------------------

    if not metric_name.startswith(
        "PasswordAuthFailureCount-"
    ):

        raise Exception(
            f"Unexpected metric name: "
            f"{metric_name}"
        )

    cluster_name = metric_name.replace(
        "PasswordAuthFailureCount-",
        ""
    )

    log_group = (
        f"/aws/rds/instance/"
        f"{cluster_name}/postgresql"
    )

    print(
        f"Cluster Name: {cluster_name}"
    )

    print(
        f"Log Group: {log_group}"
    )

    # ---------------------------------------
    # Use Exact Alarm Evaluation Window
    # ---------------------------------------

    start_date = datetime.strptime(
        reason_data["startDate"],
        "%Y-%m-%dT%H:%M:%S.%f%z"
    )

    end_date = datetime.strptime(
        reason_data["queryDate"],
        "%Y-%m-%dT%H:%M:%S.%f%z"
    )

    start_time = int(
        start_date.timestamp()
    )

    end_time = int(
        end_date.timestamp()
    )

    print(
        f"Query Start Time: "
        f"{start_time}"
    )

    print(
        f"Query End Time: "
        f"{end_time}"
    )

    # ---------------------------------------
    # Logs Insights Query
    # ---------------------------------------

    query = """
    fields @timestamp, @message
    | filter @message like /password authentication failed/
    | sort @timestamp desc
    """

    response = logs.start_query(
        logGroupName=log_group,
        startTime=start_time,
        endTime=end_time,
        queryString=query
    )

    query_id = response["queryId"]

    while True:

        result = logs.get_query_results(
            queryId=query_id
        )

        if result["status"] == "Complete":
            break

        time.sleep(1)

    print(
        f"Records Matched: "
        f"{result['statistics']['recordsMatched']}"
    )

    failures = defaultdict(int)

    sample_logs = []

    for row in result["results"]:

        message = ""

        for field in row:

            if field["field"] == "@message":

                message = field["value"]

                break

        if not message:
            continue

        user_match = re.search(
            USER_REGEX,
            message
        )

        ip_match = re.search(
            IP_REGEX,
            message
        )

        username = (
            user_match.group(1)
            if user_match
            else "Unknown"
        )

        ip_address = (
            ip_match.group(1)
            if ip_match
            else "Unknown"
        )

        failures[
            (username, ip_address)
        ] += 1

        if len(sample_logs) < 5:

            sample_logs.append(
                message
            )

    # ---------------------------------------
    # No Failures Found
    # ---------------------------------------

    if not failures:

        sns.publish(
            TopicArn=SNS_TOPIC_ARN,
            Subject=f"DEBUG - {alarm_name}",
            Message=(
                "Alarm fired but no "
                "authentication failures "
                "were found in the queried "
                "time window."
            )
        )

        return {
            "statusCode": 200
        }

    total_failures = sum(
        failures.values()
    )

    # ---------------------------------------
    # Build Email
    # ---------------------------------------

    email_body = f"""
RDS Authentication Failure Alert

AWS Account      : {event.get('accountId')}
Region           : {event.get('region')}

Cluster Name     : {cluster_name}
Alarm Name       : {alarm_name}
Metric Name      : {metric_name}

Evaluation Window : Last 5 Minutes

Alarm Configuration
----------------------------------------------------
Threshold          : {int(threshold)}
Period             : {period} seconds
Evaluation Periods : 5
Datapoints Alarm   : {datapoints_alarm}

----------------------------------------------------
Total Failures : {total_failures}

Failure Summary
----------------------------------------------------

"""

    for (
        username,
        ip
    ), count in sorted(
        failures.items(),
        key=lambda x: x[1],
        reverse=True
    ):

        email_body += f"""
Username : {username}
Client IP: {ip}
Failures : {count}

"""

    email_body += """

Sample Log Entries
----------------------------------------------------

"""

    for log in sample_logs:

        email_body += (
            log + "\n\n"
        )

    print(
        "Publishing SNS Message"
    )

    response = sns.publish(
        TopicArn=SNS_TOPIC_ARN,
        Subject=(
            f"RDS Authentication Failure Alert - "
            f"{cluster_name}"
        ),
        Message=email_body
    )

    print(
        "SNS Publish Successful"
    )

    print(response)

    return {
        "statusCode": 200
    }
